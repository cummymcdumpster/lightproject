#include <Adafruit_NeoPixel.h>

#define PIN 2          // D1 Mini pin D4
#define NUM_LEDS 8     // Your 8‑LED strip

Adafruit_NeoPixel strip(NUM_LEDS, PIN, NEO_GRB + NEO_KHZ800);

void setup() {
  strip.begin();
  strip.show(); // Initialize all pixels to 'off'
  randomSeed(analogRead(A0)); // Better randomness
}

void loop() {
  // Pick a random LED
  int idx = random(0, NUM_LEDS);

  // Pick a random color
  uint32_t color = strip.Color(
    random(0, 255),
    random(0, 255),
    random(0, 255)
  );

  // Turn everything off
  strip.clear();

  // Light the random LED
  strip.setPixelColor(idx, color);
  strip.show();

  delay(150); // Blink speed
}
